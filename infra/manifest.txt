# Co z infra/ lezi kde na serveru a jak se to nasazuje.
#
# Cte ho tools/deploy.sh (nasazeni) i tools/check-stack.sh --deep (shoda repa
# se serverem), takze nova sluzba se zapisuje jen sem. Radek je
#
#     sluzba  druh  argumenty...
#
# Soubory (porovnavaji se v --deep):
#   file     repo server          root:root 644
#   exec     repo server          root:root 700
#   unit     repo server          jako file a navic kopie do /etc/systemd/system/
#   sysunit  repo server          rovnou do /etc/systemd/system/ (jina kopie neni)
#   caddyfile repo server         {{DOMAIN}} se dosadi z CLOCK_HOST; po nasazeni reload
#   dropin   jednotka             infra/health/on-failure.conf do /etc/systemd/system/<jednotka>.d/
#
# Akce po nahrani souboru, v poradi, jak jsou zapsane:
#   restart  jednotka             systemctl restart
#   start    jednotka             systemctl start (oneshot: jeden beh hned)
#   enable   jednotka             systemctl enable --now (idempotentni)
#   run      prikaz...            libovolny prikaz pres sudo
#
# Jen s --init (prvni zavedeni sluzby):
#   user     jmeno domov          systemovy uzivatel bez shellu
#   dir      cesta vlastnik:skupina prava
#   secret   cesta vlastnik:skupina prava
#                                 prazdny soubor s hesly, jen kdyz chybi: jednotka
#                                 s EnvironmentFile= bez nej nenastartuje. Obsah
#                                 se doplni rucne (vzor *.env.example), nikdy
#                                 se neprepisuje.
#
# news.env.example tu schvalne neni: kopie v repu je opravena (Gemini), na
# serveru zustala puvodni s ANTHROPIC_API_KEY.

caddy      caddyfile infra/caddy/Caddyfile              /etc/caddy/Caddyfile
caddy      sysunit   infra/caddy/caddy.service          /etc/systemd/system/caddy.service
caddy      dropin    caddy.service
